async function mount(
  data: DiagramData,
  config: Omit<Partial<OrgHierarchyConfig<DiagramData>>, 'data'> = {},
): Promise<Mounted> {
  const container = document.createElement('div');
  container.style.width = '800px';
  container.style.height = '600px';
  Object.defineProperty(container, 'clientWidth', { configurable: true, value: 800 });
  Object.defineProperty(container, 'clientHeight', { configurable: true, value: 600 });
  document.body.appendChild(container);
  const diagram = await OrgHierarchyDiagram.create(container, {
    data,
    useWorker: false,
    ...config,
  });
  const result = { container, diagram };
  mounted.push(result);
  return result;
}

function orgRef(id: string): NodeRef {
  return { kind: 'organization', id, organizationId: id };
}

function positionRef(id: string): NodeRef {
  return { kind: 'position', id, organizationId: 'org', positionId: id };
}

function expandedOrgIds(diagram: OrgHierarchyDiagram): string[] {
  return diagram
    .getData()
    .organizations.filter((organization) => organization.collapsed !== true)
    .map((organization) => organization.id)
    .sort();
}

function hostOf(diagram: OrgHierarchyDiagram): NonNullable<DiagramInternals['host']> {
  const host = (diagram as unknown as DiagramInternals).host;
  if (!host) throw new Error('probe expected a mounted Pixi host');
  return host;
}

function pointerEvent(
  mods: Partial<{ ctrlKey: boolean; metaKey: boolean; shiftKey: boolean }> = {},
): FederatedPointerEvent {
  return {
    button: 0,
    pointerId: 1,
    clientX: 120,
    clientY: 80,
    global: { x: 120, y: 80 },
    ctrlKey: Boolean(mods.ctrlKey),
    metaKey: Boolean(mods.metaKey),
    shiftKey: Boolean(mods.shiftKey),
    stopPropagation: () => {},
    preventDefault: () => {},
    getLocalPosition: () => ({ x: 40, y: 40 }),
  } as unknown as FederatedPointerEvent;
}

function firstOrgView(diagram: OrgHierarchyDiagram): OrganizationNodeView {
  const view = hostOf(diagram).renderer.layers.organizations.children.find(
    (child): child is OrganizationNodeView => child instanceof OrganizationNodeView,
  );
  if (!view) throw new Error('probe expected a rendered organization view');
  return view;
}

function firstPersonView(diagram: OrgHierarchyDiagram): PersonNodeView {
  const view = hostOf(diagram).renderer.layers.persons.children.find(
    (child): child is PersonNodeView => child instanceof PersonNodeView,
  );
  if (!view) throw new Error('probe expected a rendered person/position view');
  return view;
}

async function flush(): Promise<void> {
  await Promise.resolve();
  await new Promise<void>((resolve) => {
    setTimeout(resolve, 0);
  });
}

async function waitUntil(predicate: () => boolean): Promise<void> {
  for (let attempt = 0; attempt < 20; attempt += 1) {
    if (predicate()) return;
    await flush();
  }
}

    diagram.openContextMenu(orgRef('branch'), { clientX: 37, clientY: 41 });
    const request = onContextMenu.mock.calls[0]?.[0] as
      | { pointer?: { clientX?: number; clientY?: number }; incomingLink?: unknown }
      | undefined;

    expect(
      request?.pointer,
      'contextMenu payload did not preserve the supplied viewport client point',
    ).toMatchObject({ clientX: 37, clientY: 41 });
    expect(
      request?.incomingLink,
      'contextMenu payload has no incomingLink for the Branch node',
    ).toEqual({ fromOrgId: 'root', toOrgId: 'branch', kind: 'administrative' });
  });

  it('row 25 — reparent emits {nodeKey, newParentKey|null}, including null detachment', async () => {
    const patches: LayoutPatch[] = [];
    const { diagram } = await mount(staffTreeData(), {
      staffCurrentOrgId: 'org',
      callbacks: { onLayoutChange: (patch) => patches.push(patch) },
    });

    await diagram.reparentPosition('b-child', 'c');
    const reparent = patches.find((patch) => patch.type === 'position-reparent');
    let detachError: string | null = null;
    try {
      await (
        diagram.reparentPosition as unknown as (
          positionId: string,
          managerId: string | null,
        ) => Promise<void>
      )('b-child', null);
    } catch (error) {
      detachError = error instanceof Error ? error.message : String(error);
    }

    expect(
      { reparent, detachError },
      'reparent payload used positionId/fromManagerId/toManagerId and null detachment was refused',
    ).toEqual({
      reparent: { nodeKey: 'b-child', newParentKey: 'c' },
      detachError: null,
    });
  });

  it('row 26 — link click emits structureId and fromDate/toDate', async () => {
    const onLinkClick = rstest.fn();
    const data = staffTreeData();
    const enrichedLine = {
      fromId: 'head',
      toId: 'a',
      kind: 'admin' as const,
      structureId: 'structure-1',
      fromDate: '2026-01-01',
      toDate: null,
    };
    data.reportLines = [enrichedLine];
    const callbacks = { onLinkClick } as UnknownCallbacks;
    const { diagram } = await mount(data, { staffCurrentOrgId: 'org', callbacks });

    hostOf(diagram).renderer.layers.edges.emit('pointertap', pointerEvent());

    expect(
      {
        eventMode: hostOf(diagram).renderer.layers.edges.eventMode,
        events: onLinkClick.mock.calls.map((call) => call[0]),
      },
      'the edge layer is non-interactive and emitted no linkClick payload with structure/date metadata',
    ).toEqual({
      eventMode: 'static',
      events: [
        {
          structureId: 'structure-1',
          fromDate: '2026-01-01',
          toDate: null,
        },
      ],
    });
  });
